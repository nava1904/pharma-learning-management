2. E-signature meanings table (FR-02-01 AC-01)
CREATE TABLE esignature_meanings ( id BIGSERIAL PRIMARY KEY, meaning_text TEXT NOT NULL, applicable_to TEXT NOT NULL, -- 'training_completion' | 'course_approval' | 'capa_closure' etc. is_active BOOLEAN DEFAULT TRUE, approved_by_id BIGINT REFERENCES pharma_users(id), approved_at TIMESTAMPTZ ); -- Seed the required meanings: INSERT INTO esignature_meanings (meaning_text, applicable_to, is_active) VALUES ('I certify that I have completed this training and will apply the content to my work', 'training_completion', TRUE), ('I approve this course version as accurate and compliant', 'course_approval', TRUE), ('I certify this CAPA is effectively closed and verified', 'capa_closure', TRUE), ('I approve this SOP document version', 'document_approval', TRUE); -- Add integrity_hash to electronic_signatures: ALTER TABLE electronic_signatures ADD COLUMN meaning_id BIGINT REFERENCES esignature_meanings(id), ADD COLUMN integrity_hash TEXT, ADD COLUMN session_id TEXT, ADD COLUMN user_agent TEXT, ADD COLUMN training_record_id BIGINT REFERENCES enrollments(id);
3. Assessment attempts table (FR-08-03 — CRITICAL)
CREATE TABLE assessment_attempts ( id BIGSERIAL PRIMARY KEY, enrollment_id BIGINT NOT NULL REFERENCES enrollments(id), assessment_id BIGINT NOT NULL REFERENCES assessments(id), attempt_number INT NOT NULL, questions_presented_json JSONB NOT NULL, -- snapshot at attempt creation answers_given_json JSONB, -- immutable after submit score_percent NUMERIC(5,2), -- computed server-side ONLY pass_mark_at_time INT NOT NULL, -- snapshotted at creation passed BOOLEAN, started_at TIMESTAMPTZ NOT NULL DEFAULT NOW(), submitted_at TIMESTAMPTZ, time_taken_seconds INT, auto_submitted BOOLEAN DEFAULT FALSE, ip_address TEXT, user_agent TEXT, session_id TEXT );
